PostgreSQL / Supabase DDL Table Relationships

Execute this script in Supabase SQL Editor to initialize full schema matching the IAS 1 Security & MAD AppDev requirements.

-- 1. Users Table (IAS Authentication & Admin User CRUD)
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    role TEXT CHECK (role IN ('Inspector', 'Supervisor', 'Engineer')) DEFAULT 'Inspector',
    status TEXT CHECK (status IN ('Active', 'Suspended', 'Locked', 'Pending Verification')) DEFAULT 'Active',
    failed_attempts INT DEFAULT 0,
    lockout_until TIMESTAMPTZ NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Inspections Table (MAD Multi-Role Transaction Cycle)
CREATE TABLE IF NOT EXISTS audit_inspections (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    area_name TEXT NOT NULL,
    severity TEXT NOT NULL,
    likelihood INT DEFAULT 1,
    severity_num INT DEFAULT 1,
    risk_score INT DEFAULT 1,
    photo_url TEXT NULL,
    estimated_cost NUMERIC(12,2) DEFAULT 0.00,
    inspector_id TEXT NULL,
    inspector_name TEXT NOT NULL,
    remarks TEXT NOT NULL,
    supervisor_status TEXT DEFAULT 'Pending Endorsement',
    supervisor_remarks TEXT NULL,
    supervisor_signature TEXT NULL,
    admin_approval_status TEXT DEFAULT 'Pending Admin Approval',
    engineer_status TEXT DEFAULT 'Pending Action',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Log Trails Module (IAS Activity Logging with Module & Result indicators)
CREATE TABLE IF NOT EXISTS log_trails (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_email TEXT NOT NULL,
    user_role TEXT NOT NULL,
    action_performed TEXT NOT NULL,
    details TEXT NOT NULL,
    ip_address TEXT DEFAULT '127.0.0.1',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. OTP Codes Table (Database Recovery & Account Verification Authentication)
CREATE TABLE IF NOT EXISTS otp_codes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email TEXT NOT NULL,
    otp_code TEXT NOT NULL,
    is_used BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. PPE Logs Table (Daily Shift Compliance Records)
CREATE TABLE IF NOT EXISTS ppe_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    worker_name TEXT NOT NULL,
    hardhat_boots BOOLEAN DEFAULT FALSE,
    reflective_vest BOOLEAN DEFAULT FALSE,
    eye_protection BOOLEAN DEFAULT FALSE,
    dust_mask BOOLEAN DEFAULT FALSE,
    logged_by TEXT NOT NULL,
    approval_status TEXT DEFAULT 'Pending',
    created_at TIMESTAMPTZ DEFAULT NOW()
);
